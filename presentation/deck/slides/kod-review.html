def review(self):
    for number in range(1, MAX_REVIEW_ROUNDS + 1):  # 2 tur
        report = self._generate("review")
        if review_verdict(report) == "APPROVE":
            break
        # REVISE: bulgular Design Agent’a gider
        self.docs["design"] = self._generate("design",
            "ARCHITECTURE REVIEW FINDINGS — revise the "
            "existing design only where these findings "
            "require it:" + report,
            base=self.docs["design"])
